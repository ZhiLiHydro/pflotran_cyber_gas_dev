# Shared 3x3x3 Richards domain for parameter regressions.
# Included via EXTERNAL_FILE from each test deck.

#=========================== discretization ===================================
GRID
  TYPE STRUCTURED
  NXYZ 3 3 3
  BOUNDS
    0.d0 0.d0 0.d0
    3.d0 3.d0 3.d0
  /
END

#=========================== material properties ==============================
MATERIAL_PROPERTY soil1
  ID 1
  POROSITY 0.25d0
  PERMEABILITY
    PERM_ISO 1.d-12
  /
  CHARACTERISTIC_CURVES cc1
END

#=========================== characteristic curves ============================
CHARACTERISTIC_CURVES cc1
  DEFAULT
END

# TIME is left to each deck (or common_domain.txt) so final time can vary.

#=========================== regions ==========================================
REGION all
  COORDINATES
    -1.d20 -1.d20 -1.d20
    1.d20 1.d20 1.d20
  /
END

#=========================== flow conditions ==================================
FLOW_CONDITION initial
  TYPE
    LIQUID_PRESSURE HYDROSTATIC
  /
  DATUM 0.d0 0.d0 3.d0
  LIQUID_PRESSURE 101325
END

#=========================== condition couplers ===============================
INITIAL_CONDITION
  FLOW_CONDITION initial
  REGION all
END

#=========================== stratigraphy couplers ============================
STRATA
  MATERIAL soil1
  REGION all
END
